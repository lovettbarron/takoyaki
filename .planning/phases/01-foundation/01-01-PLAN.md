Create `crates/takoyaki-app/src/error.rs`:
```rust
use serde::Serialize;
use thiserror::Error;

#[derive(Debug, Error, Serialize)]
pub enum AppError {
    #[error("IO error: {0}")]
    Io(String),

    #[error("Parse error: {0}")]
    Parse(String),

    #[error("Database error: {0}")]
    Db(String),

    #[error("Device error: {0}")]
    Device(String),

    #[error("Invalid path")]
    InvalidPath,
}

impl From<std::io::Error> for AppError {
    fn from(e: std::io::Error) -> Self {
        AppError::Io(e.to_string())
    }
}

impl From<rusqlite::Error> for AppError {
    fn from(e: rusqlite::Error) -> Self {
        AppError::Db(e.to_string())
    }
}

impl From<ot_parser::ParseError> for AppError {
    fn from(e: ot_parser::ParseError) -> Self {
        AppError::Parse(e.to_string())
    }
}

pub type Result<T> = std::result::Result<T, AppError>;
```
CRITICAL: AppError MUST derive serde::Serialize. Returning non-serializable errors from Tauri commands causes a runtime panic (per RESEARCH.md Pitfall 6 / PATTERNS.md).

Create `crates/takoyaki-app/src/lib.rs` with a minimal `run()` function:
```rust
mod error;

pub use error::{AppError, Result};

use tauri_specta::collect_commands;

pub fn run() {
    let builder = tauri_specta::Builder::<tauri::Wry>::new()
        .commands(collect_commands![]);

    #[cfg(debug_assertions)]
    builder
        .export(
            specta_typescript::Typescript::default(),
            "../src/bindings.ts",
        )
        .expect("Failed to export TypeScript bindings");

    tauri::Builder::default()
        .invoke_handler(builder.invoke_handler())
        .run(tauri::generate_context!())
        .expect("error while running tauri application");
}
```
NOTE: `collect_commands![]` is initially empty. Commands will be added by later plans (Plan 05 adds DB commands, Plan 07 adds device commands). The `#[cfg(debug_assertions)]` block exports TypeScript bindings to `../src/bindings.ts` relative to the crate root (which resolves to the project root's `src/bindings.ts`).
