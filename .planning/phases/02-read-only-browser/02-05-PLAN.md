**Create `src/components/health/HealthEventListener.tsx`:**

"use client" component. Renders `null` (no visible output). Listens to Tauri "health-complete" event and writes results to react-query cache.

```typescript
"use client";

import { useEffect } from "react";
import { useQueryClient } from "@tanstack/react-query";
import type { HealthCheckComplete } from "@/lib/types";

export function HealthEventListener() {
  const queryClient = useQueryClient();

  useEffect(() => {
    const cleanupFns: (() => void)[] = [];

    async function setupListeners() {
      try {
        const { listen } = await import("@tauri-apps/api/event");

        const unlisten = await listen<HealthCheckComplete>(
          "health-complete",
          (event) => {
            const { project_id } = event.payload;
            queryClient.setQueryData(["health", project_id], event.payload);
          }
        );
        cleanupFns.push(unlisten);
      } catch {
        // Not in Tauri context (SSR, browser dev)
      }
    }

    setupListeners();
    return () => cleanupFns.forEach((fn) => fn());
  }, [queryClient]);

  return null;
}
```

The `try/catch` around the dynamic import handles the case where the component runs outside Tauri (Next.js SSR or browser dev mode). The `cleanupFns` array pattern ensures all listeners are cleaned up on unmount. This exactly follows the Wallflower `tauri-event-listener.tsx` pattern.
